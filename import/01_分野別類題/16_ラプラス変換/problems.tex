\documentclass[a4paper,11pt]{article}
\usepackage{preamble}

\begin{document}

\kamokumei{類題演習 --- ラプラス変換による初期値問題}

\shijibun{次の初期値問題をラプラス変換を用いて解き、導出過程を示せ。必要があれば単位階段関数（ヘヴィサイド関数）$u(t-a)$ やデルタ関数 $\delta(t-a)$ を用いてよい。}

\shomon{問1.}{次の初期値問題を解け。}
\[
y'' + 3y' + 2y = e^{-t}, \qquad y(0) = 0,\ y'(0) = 0
\]

\shomon{問2.}{次の、デルタ関数を外力とする初期値問題を解け。}
\[
y'' + y = \delta(t - \pi), \qquad y(0) = 0,\ y'(0) = 0
\]

\shomon{問3.}{次の、ヘヴィサイド関数を外力とする初期値問題を解け。}
\[
y'' + 4y = u(t - 1), \qquad y(0) = 0,\ y'(0) = 0
\]
