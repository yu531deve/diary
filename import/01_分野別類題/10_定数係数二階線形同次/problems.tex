% 分野別類題: 定数係数2階線形同次微分方程式（y'' + ay' + by = 0）
% 特性方程式の判別（相異なる実根・重根・共役虚根）を網羅する3問。
% コンパイル: TEXINPUTS="../../../00_common:$TEXINPUTS" latexmk -xelatex problems.tex
\documentclass[a4paper,11pt]{article}
\usepackage{preamble}

\begin{document}

\kamokumei{類題演習 --- 定数係数2階線形同次微分方程式}

\shijibun{次の常微分方程式を解き、導出過程を示せ。（定数係数2階線形同次微分方程式 $y'' + ay' + by = 0$ は、特性方程式 $\lambda^{2} + a\lambda + b = 0$ の根の種類により一般解の形が定まる。）}

\shomon{問1.}{次の常微分方程式の一般解を求めよ（特性方程式が相異なる実根をもつ場合）。}
\[
y'' - 3y' + 2y = 0
\]

\shomon{問2.}{次の常微分方程式の一般解を求めよ（特性方程式が重根をもつ場合）。}
\[
y'' + 4y' + 4y = 0
\]

\shomon{問3.}{次の初期値問題を解け（特性方程式が共役虚根をもつ場合）。}
\[
y'' + 2y' + 5y = 0, \qquad y(0) = 1,\quad y'(0) = -1
\]
